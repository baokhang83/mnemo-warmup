    @Command(name = "inject", mixinStandardHelpOptions = true,
            description = "Transform rendered Kubernetes manifests from standard input.")
    static class Inject implements Callable<Integer> {
        @Spec CommandSpec spec;
        @Option(names = "--preview", description = "Report discovered workloads without emitting manifests.")
        boolean preview;
        @Option(names = "--namespace", description = "Namespace for Deployments that omit metadata.namespace.")
        String namespace;
        @Option(names = "--tools-image", description = "Tools image for the generated init containers. Default: the image whose tag is this CLI's release version.")
        String toolsImage;

        @Override public Integer call() {
            try {
                var reader = new java.io.InputStreamReader(System.in, java.nio.charset.StandardCharsets.UTF_8);
                StringBuilder input = new StringBuilder();
                char[] buffer = new char[8192];
                for (int n; (n = reader.read(buffer)) != -1;) input.append(buffer, 0, n);
                var result = OfflineInjector.inject(input.toString(), namespace, () -> toolsImage != null ? toolsImage : OfflineInjector.defaultToolsImage());
                if (preview) {
                    result.notes().forEach(spec.commandLine().getOut()::println);
                    OfflineInjector.previewImageWarnings(input.toString(), System.getenv()).forEach(spec.commandLine().getOut()::println);
                    if (result.notes().isEmpty()) spec.commandLine().getOut().println("No opted-in workloads.");
                    ManifestDiff.describe(input.toString(), result.yaml()).forEach(spec.commandLine().getOut()::println);
                } else {
                    spec.commandLine().getOut().print(result.yaml());
                    result.notes().forEach(n -> spec.commandLine().getErr().println("inject: " + n));
                }
                spec.commandLine().getOut().flush();
                return 0;
            } catch (java.io.IOException | IllegalArgumentException e) {
                spec.commandLine().getErr().println("inject: " + e.getMessage());
                return 2;
            }
        }
    }

    @Command(name = "build",
            mixinStandardHelpOptions = true,
            description = "Run a training run and produce the archive + manifest.")
    static class Build implements Callable<Integer> {

        @Spec
        CommandSpec spec;

        @Mixin
        AppOptions opts;

        @Option(
                names = "--cache-type",
                defaultValue = "auto",
                description = "auto (AOT on JDK 25+, AppCDS otherwise), appcds or aot. Default: ${DEFAULT-VALUE}.")
        String cacheType;

        @Option(names = "--image-digest", description = "Container image digest to record in the fingerprint, e.g. sha256:abc...")
        String imageDigest;

        @Option(names = "--group-readable-output", description = "Make the archive and manifest readable by the handoff group.")
        boolean groupReadableOutput;

        static void makeGroupReadable(Path file) {
            try {
                var view = Files.getFileAttributeView(file, java.nio.file.attribute.PosixFileAttributeView.class,
                        java.nio.file.LinkOption.NOFOLLOW_LINKS);
                if (view == null) throw new IllegalStateException("handoff output needs POSIX file permissions");
                var attrs = view.readAttributes();
                if (!attrs.isRegularFile()) throw new IllegalStateException("handoff output must be a regular file");
                // A new file takes the trainer's primary group, not the handoff directory's (its setgid bit is not kept).
                // The owner may move it to any group it belongs to, and the pod's fsGroup is one of them.
                Object group = Files.getAttribute(file.toAbsolutePath().getParent(), "unix:gid", java.nio.file.LinkOption.NOFOLLOW_LINKS);
                if (!group.equals(Files.getAttribute(file, "unix:gid", java.nio.file.LinkOption.NOFOLLOW_LINKS)))
                    Files.setAttribute(file, "unix:gid", group, java.nio.file.LinkOption.NOFOLLOW_LINKS);
                var permissions = java.util.EnumSet.noneOf(java.nio.file.attribute.PosixFilePermission.class);
                permissions.addAll(attrs.permissions());
                permissions.add(java.nio.file.attribute.PosixFilePermission.GROUP_READ);
                view.setPermissions(permissions);
            } catch (java.io.IOException | UnsupportedOperationException e) {
                throw new IllegalStateException("cannot make handoff output readable by its group", e);
            }
        }

        @Override
        public Integer call() {
            var out = spec.commandLine().getOut();
            var err = spec.commandLine().getErr();
            try {
                JdkInfo jdk = JdkInfo.detect(opts.javaHome, System.getenv());
                CacheStrategy strategy = CacheType.parse(cacheType).resolve(jdk);
                out.printf("Target JDK: %s %s (feature %d)%n", jdk.vendor(), jdk.version(), jdk.featureVersion());
                out.printf("Cache: %s (.%s)%n", strategy, strategy.extension());
                if (opts.jar == null) {
                    err.println("build: no --jar given, nothing to train");
                    return 0;
                }
                if (opts.timeoutSeconds <= 0) {
                    throw new IllegalArgumentException("--timeout-seconds must be positive");
                }
                var probe = opts.url == null
                        ? null
                        : new HttpProbeExitMode.Settings(
                                opts.url, Duration.ofMillis(opts.pollIntervalMillis), opts.warmupUrls, opts.warmupScript);
                ExitMode mode = ExitMode.parse(opts.exitMode, probe);
                var app = new JarExtractor().prepare(opts.jar, jdk, opts.workDir.resolve("app"));
                out.printf("Application: %s -> %s%n", app.kind(), app.runnableJar());

                Path archive = opts.workDir.resolve("app." + strategy.extension());
                out.printf("Training with exit mode %s ...%n", mode.name());
                var result = new AppTrainer()
                        .train(app, jdk, strategy, mode, archive, Duration.ofSeconds(opts.timeoutSeconds), opts.jvmArgs);
                out.printf(
                        "Archive: %s (%.1f MB) in %.1f s%n",
                        result.archive(), result.sizeBytes() / 1_048_576.0, result.elapsed().toMillis() / 1000.0);

                var manifest = ArchiveManifest.describe(app, jdk, strategy, opts.jvmArgs, System.getenv(), imageDigest)
                        .withArchive(result.archive());
                Path manifestFile = result.archive().resolveSibling(ArchiveManifest.FILE_NAME);
                manifest.write(manifestFile);
                if (groupReadableOutput) {
                    makeGroupReadable(result.archive());
                    makeGroupReadable(manifestFile);
                }
                out.printf("Manifest: %s%n", manifestFile);
                out.printf("Fingerprint: %s%n", manifest.fingerprint());
                return 0;
            } catch (IllegalArgumentException | IllegalStateException e) {
                err.println("error: " + e.getMessage());
                return 2;
            } catch (TrainingException e) {
                err.println("error: " + e.getMessage());
                return 1;
            }
        }
    }

    /**
     * The store named by {@code --store}. The login comes from {@code --registry-auth} or
     * {@code $MNEMO_WARMUP_REGISTRY_AUTH} when that file has an entry for the registry; otherwise, for an ECR
     * registry on EKS, from the pod's workload identity; otherwise the access is anonymous.
     */
    static OciArchiveStore openStore(String store, Path registryAuth, boolean plainHttp, long timeoutSeconds) {
        return openStore(store, registryAuth, plainHttp, timeoutSeconds, System.getenv());
    }

    static OciArchiveStore openStore(
            String store, Path registryAuth, boolean plainHttp, long timeoutSeconds, java.util.Map<String, String> env) {
        Path auth = registryAuth;
        if (auth == null && env.get("MNEMO_WARMUP_REGISTRY_AUTH") != null) {
            auth = Path.of(env.get("MNEMO_WARMUP_REGISTRY_AUTH"));
        }
        String host = OciArchiveStore.registryHost(store);
        RegistryCredentials credentials = auth == null ? null : RegistryCredentials.fromDockerConfig(auth, host).orElse(null);
        RegistryAuth login = credentials != null
                ? RegistryAuth.fixed(credentials)
                : AwsRegistryAuth.fromEnvironment(host, env).orElse(RegistryAuth.anonymous());
        return OciArchiveStore.withAuth(store, plainHttp, login, Duration.ofSeconds(timeoutSeconds));
    }

    /** The public keys a pod trusts, as {@code id=base64,id=base64} (docs/adr/0013-signed-archives.md). */
    static final String ENV_SIGNING_PUBLIC_KEYS = ArchiveVerifier.ENV_PUBLIC_KEYS;
    /** The file {@code keygen} wrote, mounted for the trainer. */
    static final String ENV_SIGNING_KEY_FILE = SigningKey.ENV_KEY_FILE;

    /**
     * Pull side. A key list in the environment makes a valid signature mandatory: archives that are unsigned, signed by another key
     * or altered are treated as missing. A variable that is present enforces even when it is blank, and a list in which nothing is
     * usable trusts nothing, so a typo or an empty value from a template can never switch the check off; it only costs the cache.
     * Without the variable, archives are accepted on their digests alone.
     */
    static OciArchiveStore withSignatureCheck(OciArchiveStore store, java.util.Map<String, String> env, java.util.function.Consumer<String> note) {
        String entries = env.get(ENV_SIGNING_PUBLIC_KEYS);
        if (entries == null) {
            return store;
        }
        ArchiveVerifier verifier = ArchiveVerifier.fromEntries(entries, note);
        if (verifier.isEmpty()) {
            note.accept(ENV_SIGNING_PUBLIC_KEYS + " lists no usable key, so every archive will be refused");
        }
        return store.withVerifier(verifier);
    }

    /**
     * Publish side. With a key file in the environment the store signs everything it publishes. If the file is named but cannot be used
     * (blank, missing, unreadable, malformed) this throws: asked to sign, the trainer refuses to publish unsigned.
     *
     * <p>The trainer also checks what it finds against the pods' trusted list, when that list is given: an entry any trusted key signed is
     * complete, so two trainers that use different trusted keys never overwrite each other, and a rotation moves to the new key only
     * once the old one is dropped from the list. If the trainer's own key is not on that list, every archive it published would be
     * refused by every pod, so it refuses to run. Without a list it trusts only its own key.
     */
    static OciArchiveStore withSigning(OciArchiveStore store, java.util.Map<String, String> env, boolean republishUnsigned) {
        String file = env.get(ENV_SIGNING_KEY_FILE);
        if (file == null) {
            return store;
        }
        if (file.isBlank()) {
            throw new IllegalArgumentException(ENV_SIGNING_KEY_FILE + " is set but empty; refusing to publish unsigned archives");
        }
        SigningKey key = SigningKey.fromFile(Path.of(file));
        String listed = env.get(ENV_SIGNING_PUBLIC_KEYS);
        ArchiveVerifier trusted = ArchiveVerifier.fromEntries(listed != null ? listed : key.publicKeyEntry(), n -> { });
        if (listed != null && !trusted.trusts(key.keyId())) {
            throw new IllegalArgumentException("the signing key " + key.keyId() + " is not in " + ENV_SIGNING_PUBLIC_KEYS
                    + ": every pod would refuse the archives it publishes. Add its public key to the list, or sign with a listed key");
        }
        store.withSigningKey(key);
        return republishUnsigned ? store.withVerifier(trusted) : store;
    }

    @Command(name = "keygen",
            mixinStandardHelpOptions = true,
            description = "Create an Ed25519 key pair for signing archives (docs/adr/0013-signed-archives.md). The private key goes to a "
                    + "file that is created readable by you only and never overwritten; only the public key is printed.")
    static class Keygen implements Callable<Integer> {

        @Spec
        CommandSpec spec;

        @Option(names = "--out", defaultValue = "mnemo-warmup-signing.key",
                description = "Where to write the private key file. It must not exist yet. Default: ${DEFAULT-VALUE}.")
        Path out;

        @Option(names = "--secret-name", defaultValue = "mnemo-warmup-signing",
                description = "Name used in the printed kubectl command. Default: ${DEFAULT-VALUE}.")
        String secretName;

        @Override
        public Integer call() {
            var o = spec.commandLine().getOut();
            var err = spec.commandLine().getErr();
            if (!secretName.matches("[a-z0-9]([-a-z0-9.]*[a-z0-9])?")) {
                err.println("error: --secret-name must be a valid Kubernetes name");
                return 2;
            }
            SigningKey key = SigningKey.generate();
            try {
                createPrivateFile(out, key.fileText());
            } catch (java.nio.file.FileAlreadyExistsException e) {
                err.println("error: " + out + " already exists; it is never overwritten. Choose another --out, or remove it yourself.");
                return 2;
            } catch (java.io.IOException e) {
                err.println("error: cannot write " + out + ": " + e.getMessage());
                return 2;
            }
            o.println("Key id:      " + key.keyId());
            o.println("Public key:  " + key.publicKeyEntry());
            o.println();
            o.println("The private key is in " + out + " (readable by you only). It is never printed; keep it secret and back it up.");
            o.println("Losing it does not stop any pod: archives signed with it keep working until you roll out a new key.");
            o.println();
            o.println("Give the trainer the private key as a Secret:");
            o.println("  kubectl -n <namespace> create secret generic " + secretName + " --from-file=signing.key=" + out);
            o.println("Give the pods the public key (it may be shared freely):");
            o.println("  MNEMO_WARMUP_SIGNING_PUBLIC_KEYS=" + key.publicKeyEntry());
            return 0;
        }

        /** Creates the file with owner-only permissions from the start (never readable by others, not even briefly), and refuses to overwrite. */
        static void createPrivateFile(Path file, String content) throws java.io.IOException {
            try {
                Files.createFile(file, java.nio.file.attribute.PosixFilePermissions.asFileAttribute(
                        java.nio.file.attribute.PosixFilePermissions.fromString("rw-------")));
            } catch (UnsupportedOperationException notPosix) {
                // Fail closed: a file created with inherited permissions could be readable by other users, and the promise is owner-only.
                throw new java.io.IOException("this file system cannot restrict a file to its owner; create the key on a POSIX file system "
                        + "(Linux or macOS), or in a directory only you can read, with --out");
            }
            Files.writeString(file, content, java.nio.charset.StandardCharsets.UTF_8);
        }
    }

    @Command(name = "push",
            mixinStandardHelpOptions = true,
            description = "Upload the archive and manifest produced by build to an OCI registry, tagged by fingerprint.")
    static class Push implements Callable<Integer> {

        @Spec
        CommandSpec spec;

        @Option(
                names = "--work-dir",
                defaultValue = ".mnemo-warmup",
                description = "Directory holding the archive and " + ArchiveManifest.FILE_NAME + " from build. Default: ${DEFAULT-VALUE}.")
        Path workDir;

        @Option(names = "--store", required = true, description = "Archive store: oci://host[:port]/repository.")
        String store;

        @Option(
                names = "--registry-auth",
                description = "Docker-config style JSON with the registry login (push scope). Defaults to $MNEMO_WARMUP_REGISTRY_AUTH, then anonymous.")
        Path registryAuth;

        @Option(names = "--plain-http", description = "Talk to the registry over http instead of https (local registries only).")
        boolean plainHttp;

        @Option(names = "--timeout-seconds", defaultValue = "1800", description = "Per request. Default: ${DEFAULT-VALUE}.")
        long timeoutSeconds;

        @Override
        public Integer call() {
            var out = spec.commandLine().getOut();
            var err = spec.commandLine().getErr();
            try {
                Path manifestFile = workDir.resolve(ArchiveManifest.FILE_NAME);
                if (!Files.isRegularFile(manifestFile)) {
                    err.println("error: no " + manifestFile + " (run build first)");
                    return 2;
                }
                ArchiveManifest manifest = ArchiveManifest.read(manifestFile);
                if (manifest.archive() == null) {
                    err.println("error: " + manifestFile + " does not describe an archive");
                    return 2;
                }
                Path archive = workDir.resolve(manifest.archive().file());
                if (!Files.isRegularFile(archive)) {
                    err.println("error: archive " + archive + " not found");
                    return 2;
                }
                if (timeoutSeconds <= 0) {
                    throw new IllegalArgumentException("--timeout-seconds must be positive");
                }
                var target = withSigning(openStore(store, registryAuth, plainHttp, timeoutSeconds), System.getenv(), false);
                out.printf("Uploading %s (%.1f MB) to %s%n", archive, manifest.archive().sizeBytes() / 1_048_576.0, store);
                target.put(manifest, archive);
                out.printf("Stored fingerprint %s as tag %s%n", manifest.fingerprint(), OciArchiveStore.tagFor(manifest.fingerprint()));
                return 0;
            } catch (IllegalArgumentException | IllegalStateException e) {
                err.println("error: " + e.getMessage());
                return 2;
            } catch (StoreException e) {
                err.println("error: " + e.getMessage());
                return 1;
            }
        }
    }

    @Command(name = "fetch",
            mixinStandardHelpOptions = true,
            description = "Init container: look up the archive for this JDK, classpath and options, verify it and write it with a flags file. "
                    + "Always exits 0 unless --strict, so a missing or bad archive never stops the pod.")
    static class Fetch implements Callable<Integer> {

        @Spec
        CommandSpec spec;

        @Mixin
        AppOptions opts;

        @Option(names = "--store", description = "Archive store: oci://host[:port]/repository.")
        String store;

        @Option(
                names = "--out",
                defaultValue = ".mnemo-warmup/cache",
                description = "Directory that receives the archive, manifest and " + ArchiveFetcher.FLAGS_FILE + ". Default: ${DEFAULT-VALUE}.")
        Path out;

        @Option(
                names = "--cache-type",
                defaultValue = "auto",
                description = "auto (try AOT on JDK 25+, then AppCDS), appcds or aot. Default: ${DEFAULT-VALUE}.")
        String cacheType;

        @Option(names = "--image-digest", description = "Container image digest, as given to build.")
        String imageDigest;

        @Option(
                names = "--registry-auth",
                description = "Docker-config style JSON with the registry login (pull scope). Defaults to $MNEMO_WARMUP_REGISTRY_AUTH, then anonymous.")
        Path registryAuth;

        @Option(names = "--plain-http", description = "Talk to the registry over http instead of https (local registries only).")
        boolean plainHttp;

        @Option(names = "--store-timeout-seconds", defaultValue = "120", description = "Per registry request and download stall. Default: ${DEFAULT-VALUE}.")
        long storeTimeoutSeconds;

        @Option(names = "--strict", description = "Exit 1 on a store problem and 2 on bad arguments instead of 0. For CI and debugging only.")
        boolean strict;

        @Override
        public Integer call() {
            var o = spec.commandLine().getOut();
            var e = spec.commandLine().getErr();
            try {
                ArchiveFetcher.clear(out);
            } catch (RuntimeException ex) {
                return fail(e, "warning: " + ex.getMessage(), 1);
            }
            try {
                if (store == null || opts.jar == null) {
                    throw new IllegalArgumentException("fetch needs --store and --jar");
                }
                if (storeTimeoutSeconds <= 0) {
                    throw new IllegalArgumentException("--store-timeout-seconds must be positive");
                }
                var timer = io.github.baokhang83.mnemo.warmup.core.PhaseTimer.from(System.getenv());
                JdkInfo jdk = JdkInfo.detect(opts.javaHome, System.getenv());
                timer.lap("jdk-detect");
                List<CacheStrategy> order = candidates(jdk);
                // Extraction (needed to hash a Spring Boot fat JAR's libraries) goes to a fresh directory every
                // time, because JarExtractor refuses a non-empty destination and fetch must be repeatable,
                // for example after an init container restart on a shared volume. The fingerprint uses entry
                // names and content, never the extraction path.
                Path scratch;
                try {
                    Files.createDirectories(opts.workDir);
                    scratch = Files.createTempDirectory(opts.workDir, "fetch-");
                } catch (java.io.IOException io) {
                    throw new IllegalStateException("Cannot create scratch space in " + opts.workDir + ": " + io.getMessage(), io);
                }
                List<ArchiveManifest> live = new ArrayList<>();
                try {
                    var app = new JarExtractor().prepare(opts.jar, jdk, scratch.resolve("app"));
                    for (CacheStrategy strategy : order) {
                        live.add(ArchiveManifest.describe(app, jdk, strategy, opts.jvmArgs, System.getenv(), imageDigest));
                    }
                } finally {
                    deleteTree(scratch);
                }
                timer.lap("extract-and-hash-classpath");
                // The total deadline starts here, after the local hashing: it bounds what the registry can make the pod wait.
                var archiveStore = withSignatureCheck(
                        openStore(store, registryAuth, plainHttp, storeTimeoutSeconds)
                                .withFetchLimits(FetchLimits.fromEnvironment(System.getenv(), note -> e.println("fetch: " + note))),
                        System.getenv(), note -> e.println("fetch: " + note));
                var result = new ArchiveFetcher().fetch(archiveStore, live, out);
                timer.lap("store-lookup-download-verify");
                result.notes().forEach(n -> e.println("fetch: " + n));
                timer.summary().ifPresent(t -> e.println("fetch: " + t));
                if (result.outcome() == ArchiveFetcher.Outcome.HIT) {
                    o.printf("Archive: %s (%s), fingerprint %s%n", result.archive(), result.cacheType(), result.fingerprint());
                    o.printf("Flags: %s%n", result.flagsFile());
                    return 0;
                }
                o.println("No usable archive; the application will start without one.");
                return strict ? 1 : 0;
            } catch (IllegalArgumentException | IllegalStateException ex) {
                return fail(e, "warning: " + ex.getMessage(), 2);
            } catch (RuntimeException ex) {
                return fail(e, "warning: " + ex.getMessage(), 1);
            }
        }

        private static void deleteTree(Path root) {
            try (var paths = Files.walk(root)) {
                paths.sorted(java.util.Comparator.reverseOrder()).forEach(p -> p.toFile().delete());
            } catch (java.io.IOException | RuntimeException ignored) {
                // best effort: the directory is scratch space
            }
        }

        /**
         * Picocli rejects bad options before call() runs, so this is the fail-open path for them: it
         * withdraws any stale flags (best effort, from --out if it can be found) and exits 0 unless
         * --strict was given.
         */
        static int failOpenOnParseError(CommandLine.ParameterException ex, String[] args) {
            var err = ex.getCommandLine().getErr();
            boolean strict = java.util.Arrays.asList(args).contains("--strict");
            err.println("warning: " + ex.getMessage());
            Path outDir = Path.of(".mnemo-warmup/cache");
            for (int i = 0; i < args.length; i++) {
                if (args[i].equals("--out") && i + 1 < args.length) {
                    outDir = Path.of(args[i + 1]);
                } else if (args[i].startsWith("--out=")) {
                    outDir = Path.of(args[i].substring("--out=".length()));
                }
            }
            try {
                ArchiveFetcher.clear(outDir);
            } catch (RuntimeException clearFailure) {
                err.println("warning: " + clearFailure.getMessage());
            }
            err.println("fetch: continuing without an archive" + (strict ? " (--strict: failing)" : ""));
            return strict ? 2 : 0;
        }

        static int failOpenOnUnexpectedError(Exception ex, CommandLine commandLine) {
            var err = commandLine.getErr();
            boolean strict = commandLine.getCommand() instanceof Fetch f && f.strict;
            err.println("warning: unexpected error: " + ex);
            err.println("fetch: continuing without an archive" + (strict ? " (--strict: failing)" : ""));
            return strict ? 1 : 0;
        }

        private List<CacheStrategy> candidates(JdkInfo jdk) {
            return switch (CacheType.parse(cacheType)) {
                case AUTO -> jdk.featureVersion() >= CacheStrategy.LEYDEN_MIN_FEATURE
                        ? List.of(CacheStrategy.LEYDEN_AOT, CacheStrategy.APP_CDS)
                        : List.of(CacheStrategy.APP_CDS);
                case APPCDS -> List.of(CacheStrategy.APP_CDS);
                case AOT -> List.of(CacheType.AOT.resolve(jdk));
            };
        }

        private int fail(java.io.PrintWriter err, String message, int strictExit) {
            err.println(message);
            err.println("fetch: continuing without an archive" + (strict ? " (--strict: failing)" : ""));
            return strict ? strictExit : 0;
        }
    }

    @Command(name = "run",
            mixinStandardHelpOptions = true,
            description = "Application entrypoint: start 'java ... -jar app.jar' with the fetched archive if it still matches this exact "
                    + "launch, otherwise exactly as given. Put the command after --.",
            footer = "%nExample: mnemo-warmup run --print -- java -Xmx512m -jar /app/app.jar")
    static class Run implements Callable<Integer> {

        @Spec
        CommandSpec spec;

        @Option(
                names = "--cache-dir",
                defaultValue = "${MNEMO_WARMUP_CACHE_DIR:-.mnemo-warmup/cache}",
                description = "Directory written by fetch --out. Default: $MNEMO_WARMUP_CACHE_DIR, then ${DEFAULT-VALUE}.")
        Path cacheDir;

        @Option(
                names = "--image-digest",
                defaultValue = "${MNEMO_WARMUP_IMAGE_DIGEST:-}",
                description = "Container image digest, as given to build. Default: $MNEMO_WARMUP_IMAGE_DIGEST.")
        String imageDigest;

        @Option(names = "--print", description = "Print the final command, shell-quoted on one line, and exit 0, so a shell can exec it "
                + "(scripts/mnemo-warmup-exec). The JVM is then PID 1 with no wrapper process.")
        boolean print;

        @Option(names = "--format", defaultValue = "shell", description = "--print: shell (one shell-quoted line, for eval) or nul "
                + "(each argument followed by a NUL byte: exact argv with no quoting rules, for the native launcher). Default: ${DEFAULT-VALUE}.")
        String format;

        @Option(names = "--spawn", description = "Start the command as a child process, forward SIGTERM to it and exit with its status. "
                + "Keeps this JVM resident (about 40-60 MB); for images without /bin/sh. This is the default without --print.")
        boolean spawn;

        @Option(
                names = "--stop-timeout-seconds",
                defaultValue = "25",
                description = "--spawn: after SIGTERM, how long to wait for the command to exit before killing it. "
                        + "If process startup is still in progress, shutdown can take up to twice this value. "
                        + "Keep that within the pod's terminationGracePeriodSeconds. Default: ${DEFAULT-VALUE}.")
        long stopTimeoutSeconds;

        @Parameters(arity = "1..*", paramLabel = "COMMAND", description = "The command to start, after --.")
        List<String> command = new ArrayList<>();

        @Override
        public Integer call() {
            var err = spec.commandLine().getErr();
            if (print && spawn) {
                err.println("error: --print and --spawn are exclusive");
                return 2;
            }
            String digest = imageDigest == null || imageDigest.isBlank() ? null : imageDigest;
            var plan = new RunPlanner().plan(command, cacheDir, System.getenv(), digest);
            plan.notes().forEach(n -> err.println("run: " + n));
            if (!"shell".equals(format) && !"nul".equals(format)) {
                err.println("error: --format must be shell or nul");
                return 2;
            }
            return launch(spec.commandLine(), plan.command(), print, "nul".equals(format), Duration.ofSeconds(Math.max(1, stopTimeoutSeconds)));
        }

        static int launch(CommandLine cmd, List<String> command, boolean print, boolean nul, Duration stopTimeout) {
            if (print) {
                if (nul) {
                    cmd.getOut().print(nulSeparated(command));
                    cmd.getOut().flush();
                } else {
                    cmd.getOut().println(shellQuote(command));
                }
                return 0;
            }
            // The hook is registered BEFORE the child starts: a SIGTERM that arrives while the child is
            // still coming up (or while this JVM is still linking the hook) must not be lost, or this JVM
            // would exit at once and leave the child running with no grace period.
            var child = new java.util.concurrent.CompletableFuture<Process>();
            Thread forwarder = new Thread(() -> {
                stopWhenPublished(child, stopTimeout);
            }, "run-signal-forwarder");
            Runtime.getRuntime().addShutdownHook(forwarder);
            try {
                Process process = new ProcessBuilder(command).inheritIO().start();
                child.complete(process);
                try {
                    return process.waitFor();
                } catch (InterruptedException e) {
                    process.destroy();
                    Thread.currentThread().interrupt();
                    return 130;
                }
            } catch (java.io.IOException e) {
                child.completeExceptionally(e);
                cmd.getErr().println("error: cannot start " + command.get(0) + ": " + e.getMessage());
                return 127;
            } finally {
                try {
                    Runtime.getRuntime().removeShutdownHook(forwarder);
                } catch (IllegalStateException ignored) {
                    // JVM is already shutting down
                }
            }
        }

        static void stopWhenPublished(java.util.concurrent.CompletableFuture<Process> child, Duration timeout) {
            try {
                stop(child.get(timeout.toMillis(), java.util.concurrent.TimeUnit.MILLISECONDS), timeout);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            } catch (java.util.concurrent.ExecutionException | java.util.concurrent.TimeoutException ignored) {
                // Starting failed or did not finish within the bounded shutdown wait.
            }
        }

        /**
         * Forwards SIGTERM to the child and WAITS for it. When this JVM is PID 1 it must not exit while the
         * child is still running its shutdown hooks, or the container runtime kills the child at once
         * instead of giving it the grace period. Falls back to SIGKILL after {@code timeout}.
         *
         * @return whether the child exited on its own within the timeout
         */
        static boolean stop(Process process, Duration timeout) {
            process.destroy();
            try {
                if (process.waitFor(timeout.toMillis(), java.util.concurrent.TimeUnit.MILLISECONDS)) {
                    return true;
                }
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
            process.destroyForcibly();
            return false;
        }

        /** Each argument followed by a NUL byte, which no argument can contain: exact argv, nothing to unquote. */
        static String nulSeparated(List<String> command) {
            var sb = new StringBuilder();
            for (String arg : command) {
                sb.append(arg).append('\0');
            }
            return sb.toString();
        }

        /** {@code 'a' 'b c' 'it'\''s'}: safe for {@code eval} in any POSIX shell. */
        static String shellQuote(List<String> command) {
            var sb = new StringBuilder();
            for (String arg : command) {
                if (sb.length() > 0) {
                    sb.append(' ');
                }
                sb.append('\'').append(arg.replace("'", "'\\''")).append('\'');
            }
            return sb.toString();
        }

        /**
         * A bad option to {@code run} must not stop the application: whatever follows {@code --} is still
         * the command to start, so start it unchanged (or print it, with --print).
         */
        static int failOpenOnParseError(CommandLine.ParameterException ex, String[] args) {
            var cmd = ex.getCommandLine();
            int dashes = java.util.Arrays.asList(args).indexOf("--");
            if (dashes < 0 || dashes == args.length - 1) {
                cmd.getErr().println("error: " + ex.getMessage());
                cmd.getErr().println("usage: mnemo-warmup run [--print|--spawn] -- <command>");
                return 2;
            }
            cmd.getErr().println("warning: " + ex.getMessage() + "; starting the command unchanged");
            List<String> original = List.of(args).subList(dashes + 1, args.length);
            List<String> options = java.util.Arrays.asList(args).subList(0, dashes);
            boolean print = options.contains("--print");
            boolean nul = options.contains("--format=nul") || (options.indexOf("--format") >= 0
                    && options.indexOf("--format") + 1 < options.size() && "nul".equals(options.get(options.indexOf("--format") + 1)));
            return launch(cmd, original, print, nul, Duration.ofSeconds(25));
        }
    }

    @Command(name = "train",
            mixinStandardHelpOptions = true,
            description = "Trainer (a CronJob in the application image): if the store has no archive for this launch, "
                    + "take a lock, train one and upload it. Exits 0 for every outcome, including failures: the next scheduled run retries.")
    static class Train implements Callable<Integer> {

        @Spec
        CommandSpec spec;

        @Mixin
        AppOptions opts;

        @Option(names = "--store", required = true, description = "Archive store: oci://host[:port]/repository.")
        String store;

        @Option(names = "--cache-type", defaultValue = "auto", description = "auto, appcds or aot. Default: ${DEFAULT-VALUE}.")
        String cacheType;

        @Option(names = "--image-digest", description = "Container image digest; the same value fetch and run use.")
        String imageDigest;

        @Option(names = "--registry-auth", description = "Docker-config style JSON with the registry login (push scope). "
                + "Defaults to $MNEMO_WARMUP_REGISTRY_AUTH, then anonymous.")
        Path registryAuth;

        @Option(names = "--plain-http", description = "Talk to the registry over http instead of https (local registries only).")
        boolean plainHttp;

        @Option(names = "--store-timeout-seconds", defaultValue = "1800", description = "Per registry request. Default: ${DEFAULT-VALUE}.")
        long storeTimeoutSeconds;

        @Option(names = "--lock", defaultValue = "lease", description = "lease (a Kubernetes Lease named after the fingerprint) or none. Default: ${DEFAULT-VALUE}.")
        String lock;

        @Option(names = "--lease-seconds", defaultValue = "1800", description = "How long a crashed trainer's lease blocks others. Default: ${DEFAULT-VALUE}.")
        long leaseSeconds;

        @Option(names = "--strict", description = "Exit 1 when the outcome is a failure. For CI and debugging.")
        boolean strict;

        @Override
        public Integer call() {
            var out = spec.commandLine().getOut();
            var err = spec.commandLine().getErr();
            try {
                if (opts.jar == null) {
                    throw new IllegalArgumentException("train needs --jar");
                }
                if (leaseSeconds < 3 || storeTimeoutSeconds <= 0 || opts.timeoutSeconds <= 0) {
                    throw new IllegalArgumentException("--lease-seconds must be at least 3 and the timeouts positive");
                }
                if (!lock.equals("lease") && !lock.equals("none")) {
                    throw new IllegalArgumentException("--lock must be lease or none");
                }
                JdkInfo jdk = JdkInfo.detect(opts.javaHome, System.getenv());
                CacheStrategy strategy = CacheType.parse(cacheType).resolve(jdk);
                if (JarExtractor.needsExtraction(opts.jar)) {
                    throw new IllegalArgumentException("train needs the extracted layout: extract the Spring Boot JAR at image build time "
                            + "(java -Djarmode=tools -jar app.jar extract) so the trainer and the app run the same files (ADR 0003)");
                }
                var probe = opts.url == null
                        ? null
                        : new HttpProbeExitMode.Settings(
                                opts.url, Duration.ofMillis(opts.pollIntervalMillis), opts.warmupUrls, opts.warmupScript);
                ExitMode mode = ExitMode.parse(opts.exitMode, probe);
                // the extracted layout is used as it is; prepare() reports its framework (Spring Boot or not), which is
                // what spring-refresh checks
                var app = new JarExtractor().prepare(opts.jar, jdk, opts.workDir.resolve("unused"));
                ArchiveManifest live = ArchiveManifest.describe(app, jdk, strategy, opts.jvmArgs, System.getenv(), imageDigest);

                Path archive = opts.workDir.resolve("app." + strategy.extension());
                java.util.concurrent.Callable<Path> training = () -> {
                    java.nio.file.Files.createDirectories(opts.workDir);
                    java.nio.file.Files.deleteIfExists(archive);
                    out.printf("Training %s with exit mode %s ...%n", strategy, mode.name());
                    return new AppTrainer()
                            .train(app, jdk, strategy, mode, archive, Duration.ofSeconds(opts.timeoutSeconds), opts.jvmArgs)
                            .archive();
                };

                TrainingOrchestrator.Outcome outcome;
                try {
                    TrainingLock trainingLock = lock.equals("none") ? TrainingLock.none() : KubernetesLeaseLock.inCluster(Duration.ofSeconds(leaseSeconds));
                    outcome = new TrainingOrchestrator().ensure(
                            live, withSigning(openStore(store, registryAuth, plainHttp, storeTimeoutSeconds), System.getenv(), true), trainingLock, training);
                } catch (StoreException e) {
                    outcome = new TrainingOrchestrator.Outcome(TrainingOrchestrator.Status.FAILED, live.fingerprint(), e.getMessage());
                }
                (outcome.status() == TrainingOrchestrator.Status.FAILED ? err : out)
                        .printf("train: %s: %s (%s)%n", outcome.status(), outcome.detail(), outcome.fingerprint());
                return outcome.status() == TrainingOrchestrator.Status.FAILED && strict ? 1 : 0;
            } catch (IllegalArgumentException | IllegalStateException e) {
                err.println("error: " + e.getMessage());
                return 2;
            }
        }
    }

        String trainingTemplateSha256;
        @Option(names = "--allowed-secret", description = "A Secret the training pod may reference (repeatable); any other is refused.")
        List<String> allowedSecrets = new java.util.ArrayList<>();
        @Option(names = "--training-service-account", required = true) String trainingServiceAccount;
        @Option(names = "--handoff-claim", required = true) String claim;
        @Option(names = "--handoff-root", required = true) Path claimRoot;
        @Option(names = "--signing-secret-name", required = true) String signingSecret;
        @Option(names = "--push-secret-name") String pushSecret;

        @Option(names = "--lease-seconds", defaultValue = "1800") long leaseSeconds;
        @Option(names = "--max-archive-bytes", defaultValue = "1073741824") long maxArchiveBytes;

        /**
         * Whoever can write the template's ConfigMap would decide what this pod-creating coordinator creates, and ConfigMap write access
         * is often granted more widely than pod creation. The digest comes from the CronJob, whose edit needs about pod-create power.
         */
        static java.util.Map<String, Object> readTemplate(Path path, String expectedSha256) throws java.io.IOException {
            if (expectedSha256 == null || !expectedSha256.matches("[0-9a-f]{64}"))
                throw new IllegalArgumentException("--training-template-sha256 must be a lowercase hex SHA-256");
            try (var in = Files.newInputStream(path)) {
                byte[] bytes = in.readNBytes((1 << 20) + 1);
                if (bytes.length > (1 << 20)) throw new IllegalArgumentException("training template is too large");
                String actual;
                try {
                    actual = java.util.HexFormat.of().formatHex(java.security.MessageDigest.getInstance("SHA-256").digest(bytes));
                } catch (java.security.NoSuchAlgorithmException e) { throw new IllegalStateException(e); }
                if (!java.security.MessageDigest.isEqual(actual.getBytes(java.nio.charset.StandardCharsets.US_ASCII),
                        expectedSha256.getBytes(java.nio.charset.StandardCharsets.US_ASCII)))
                    throw new IllegalArgumentException("training template does not match --training-template-sha256 (was its ConfigMap changed?)");
                var documents = ManifestStream.parse(new String(bytes, java.nio.charset.StandardCharsets.UTF_8));
                if (documents.size() != 1) throw new IllegalArgumentException("training template must contain one pod spec");
                return documents.get(0);
            }
        }

        static java.util.Set<String> protectedSecretNames(String signing, String push, boolean hasRegistryAuth) {
            if (!validSecretName(signing)) throw new IllegalArgumentException("a valid signing Secret name is required");
            if (hasRegistryAuth && (push == null || push.isBlank())) {
                throw new IllegalArgumentException("the registry push Secret name is required when registry credentials are mounted");
            }
            var names = new java.util.HashSet<String>();
            names.add(signing);
            if (push != null && !push.isBlank()) {
                if (!validSecretName(push)) throw new IllegalArgumentException("invalid registry push Secret name");
                names.add(push);
            }
            return java.util.Set.copyOf(names);
        }

        static java.util.Set<String> allowedSecretNames(List<String> names) {
            for (String name : names) if (!validSecretName(name)) throw new IllegalArgumentException("invalid allowed Secret name: " + name);
            return java.util.Set.copyOf(names);
        }

        /** The coordinator's own pod, from the downward API, to own the training pods it creates; null outside such a pod. */
        static IsolatedTrainingPodFactory.Owner owner(java.util.Map<String, String> env) {
            String name = env.get("POD_NAME"), uid = env.get("POD_UID");
            if ((name == null || name.isBlank()) && (uid == null || uid.isBlank())) return null;
            return new IsolatedTrainingPodFactory.Owner(name, uid);
        }

        static void requireDistinctServiceAccounts(String coordinator, String training) {
            if (coordinator == null || coordinator.isBlank() || training == null || training.isBlank()
                    || coordinator.equals(training)) {
                throw new IllegalArgumentException("training and coordinator service accounts must be different");
            }
        }

        static void requireMatchingHandoffGroup(Path root, java.util.Map<String, Object> template) throws java.io.IOException {
            Object raw = template.get("securityContext");
            Object fsGroup = raw instanceof java.util.Map<?, ?> sc ? sc.get("fsGroup") : null;
            Object group = Files.getAttribute(root, "unix:gid", java.nio.file.LinkOption.NOFOLLOW_LINKS);
            if (!(fsGroup instanceof Number n) || !(group instanceof Number g)
                    || n.longValue() <= 0 || n.longValue() != g.longValue()) {
                throw new IllegalArgumentException("training fsGroup must match the handoff claim root group");
            }
        }

        private static boolean validSecretName(String name) {
            if (name == null || name.length() > 253 || name.isBlank()) return false;
            for (String label : name.split("\\.", -1)) {
                if (label.length() > 63 || !label.matches("[a-z0-9]([-a-z0-9]*[a-z0-9])?")) return false;
            }
            return true;
        }

        @Override public Integer call() {
            var out = spec.commandLine().getOut();
            var err = spec.commandLine().getErr();
            try {
                if (opts.jar == null) throw new IllegalArgumentException("coordinate-isolated needs --jar");
                if (trainingTimeout < 1 || trainingTimeout > 86400 || storeTimeout <= 0 || leaseSeconds < 3 || maxArchiveBytes <= 0)
                    throw new IllegalArgumentException("invalid training timeout, store timeout, lease duration or archive limit");
                String key = System.getenv(ENV_SIGNING_KEY_FILE);
                if (key == null || key.isBlank()) throw new IllegalArgumentException("coordinate-isolated requires a signing key");
                String namespace = System.getenv("POD_NAMESPACE");
                String coordinatorAccount = IsolatedTrainingPodRunner.coordinatorServiceAccount(namespace);
                requireDistinctServiceAccounts(coordinatorAccount, trainingServiceAccount);
                if (JarExtractor.needsExtraction(opts.jar)) throw new IllegalArgumentException("coordinate-isolated requires a pre-extracted application layout");
                JdkInfo jdk = isolatedPublisherJdk(opts.javaHome);
                CacheStrategy strategy = CacheType.parse(cacheType).resolve(jdk);
                var app = new JarExtractor().prepare(opts.jar, jdk, opts.workDir.resolve("unused"));
                ArchiveManifest live = ArchiveManifest.describe(app, jdk, strategy, opts.jvmArgs,
                        IsolatedPublish.targetJavaEnvironment(System.getenv()), imageDigest);
                var target = withSigning(openStore(store, registryAuth, plainHttp, storeTimeout), System.getenv(), true);
                var template = readTemplate(trainingTemplate, trainingTemplateSha256);
                requireMatchingHandoffGroup(claimRoot, template);
                var protectedSecrets = protectedSecretNames(signingSecret, pushSecret,
                        registryAuth != null || System.getenv("MNEMO_WARMUP_REGISTRY_AUTH") != null);
                var allowed = allowedSecretNames(allowedSecrets);
                var owner = owner(System.getenv());
                // Refuse a bad template before a handoff exists: a refusal later would look like an uncertain run.
                IsolatedTrainingPodFactory.build(template, namespace, "mnemo-train-check", trainingServiceAccount, claim,
                        "mnemo-run-check", protectedSecrets, allowed, trainingTimeout, owner);
                var runner = IsolatedTrainingPodRunner.inCluster();
                var result = new IsolatedTrainingCoordinator().run(live, target, claimRoot, handoff -> {
                    String name = "mnemo-train-" + java.util.UUID.randomUUID().toString().replace("-", "");
                    var pod = IsolatedTrainingPodFactory.build(template, namespace, name, trainingServiceAccount, claim,
                            handoff, protectedSecrets, allowed, trainingTimeout, owner);
                    return runner.run(namespace, pod, Duration.ofSeconds(trainingTimeout), Duration.ofSeconds(1));
                }, dir -> new IsolatedArchivePublisher().publish(live, target, KubernetesLeaseLock.inCluster(Duration.ofSeconds(leaseSeconds)),
                        dir.resolve("app." + strategy.extension()), maxArchiveBytes));
                (result.status() == IsolatedTrainingCoordinator.Status.FAILED ? err : out)
                        .printf("coordinate-isolated: %s: %s%n", result.status(), result.detail());
                if (result.retainedHandoff() != null) err.println("coordinate-isolated: retained handoff " + result.retainedHandoff());
                return result.status() == IsolatedTrainingCoordinator.Status.FAILED ? 1 : 0;
            } catch (java.io.IOException | IllegalArgumentException | IllegalStateException | StoreException e) {
                err.println("coordinate-isolated: " + e.getMessage());
                return 1;
            }
        }
    }

    @Command(name = "bench",
            mixinStandardHelpOptions = true,
            description = "Compare cold starts with no archive, AppCDS and AOT, and print a Markdown table.")
    static class Bench implements Callable<Integer> {

        /** Fixed so every variant is timed with the same resolution: a reading can be late by at most this much. */
        private static final Duration MEASURE_POLL = Duration.ofMillis(20);

        @Spec
        CommandSpec spec;

        @Mixin
        AppOptions opts;

        @Option(names = "--runs", defaultValue = "3", description = "Timed cold starts per variant. Default: ${DEFAULT-VALUE}.")
        int runs;

        @Option(
                names = "--warmups",
                defaultValue = "1",
                description = "Starts per variant that are run first and thrown away. Default: ${DEFAULT-VALUE}.")
        int warmups;

        @Override
        public Integer call() {
            var out = spec.commandLine().getOut();
            var err = spec.commandLine().getErr();
            try {
                if (opts.jar == null) {
                    throw new IllegalArgumentException("bench needs --jar");
                }
                if (opts.timeoutSeconds <= 0 || runs < 1 || warmups < 0) {
                    throw new IllegalArgumentException("--timeout-seconds and --runs must be positive and --warmups not negative");
                }
                JdkInfo jdk = JdkInfo.detect(opts.javaHome, System.getenv());
                var trainProbe = opts.url == null
                        ? null
                        : new HttpProbeExitMode.Settings(
                                opts.url, Duration.ofMillis(opts.pollIntervalMillis), opts.warmupUrls, opts.warmupScript);
                ExitMode trainMode = ExitMode.parse(opts.exitMode, trainProbe);
                // Same mode for measuring, but no warmup traffic: that would be counted as startup.
                ExitMode measureMode = opts.url == null
                        ? trainMode
                        : ExitMode.parse(opts.exitMode, HttpProbeExitMode.Settings.of(opts.url)
                                .withPollInterval(MEASURE_POLL));
                var app = new JarExtractor().prepare(opts.jar, jdk, opts.workDir.resolve("app"));
                Duration timeout = Duration.ofSeconds(opts.timeoutSeconds);

                var variants = new ArrayList<StartupBench.Variant>();
                variants.add(StartupBench.Variant.none());
                for (CacheStrategy strategy : strategiesFor(jdk)) {
                    Path archive = opts.workDir.resolve("bench." + strategy.extension());
                    err.printf("Training %s ...%n", strategy);
                    new AppTrainer().train(app, jdk, strategy, trainMode, archive, timeout, opts.jvmArgs);
                    variants.add(StartupBench.Variant.of(label(strategy), strategy, archive));
                }

                var entries = new ArrayList<BenchReport.Entry>();
                for (var variant : variants) {
                    err.printf("Measuring %s (%d warmup + %d timed starts) ...%n", variant.name(), warmups, runs);
                    var samples = new StartupBench().measure(
                            jdk, variant, app, measureMode, opts.jvmArgs, timeout, warmups, runs,
                            opts.workDir.resolve("bench-" + variant.name() + ".log"));
                    entries.add(new BenchReport.Entry(variant.name(), samples));
                }
                out.printf("JDK %s %s, %d timed starts per variant (median):%n%n", jdk.vendor(), jdk.version(), runs);
                out.println(BenchReport.markdown(entries));
                return 0;
            } catch (IllegalArgumentException | IllegalStateException e) {
                err.println("error: " + e.getMessage());
                return 2;
            } catch (TrainingException e) {
                err.println("error: " + e.getMessage());
                return 1;
            }
        }

        /** AppCDS is available on every supported JDK; AOT only where the single-step flags exist. */
        static List<CacheStrategy> strategiesFor(JdkInfo jdk) {
            var list = new ArrayList<CacheStrategy>();
            list.add(CacheStrategy.APP_CDS);
            if (jdk.featureVersion() >= CacheStrategy.LEYDEN_MIN_FEATURE) {
                list.add(CacheStrategy.LEYDEN_AOT);
            }
            return list;
        }

        private static String label(CacheStrategy strategy) {
            return strategy == CacheStrategy.APP_CDS ? "appcds" : "aot";
        }
    }

    static CommandLine commandLine() {
        var cmd = new CommandLine(new Main());
        // fetch runs as an init container: a bad option or an unexpected error must not stop the pod (§3).
        var defaultParseHandler = cmd.getParameterExceptionHandler();
        cmd.setParameterExceptionHandler((ex, args) -> switch (ex.getCommandLine().getCommandName()) {
            case "fetch" -> Fetch.failOpenOnParseError(ex, args);
            case "run" -> Run.failOpenOnParseError(ex, args);
            default -> defaultParseHandler.handleParseException(ex, args);
        });
        var defaultExecutionHandler = cmd.getExecutionExceptionHandler();
        cmd.setExecutionExceptionHandler((ex, commandLine, parseResult) -> "fetch".equals(commandLine.getCommandName())
                ? Fetch.failOpenOnUnexpectedError(ex, commandLine)
                : defaultExecutionHandler.handleExecutionException(ex, commandLine, parseResult));
        return cmd;
    }

    public static void main(String[] args) {
        System.exit(exitCode(args, () -> commandLine().execute(args)));
    }

    /**
     * Runs a command and returns its exit code. {@code fetch} runs as an init container: even running out of memory (or stack) on a
     * hostile response must not keep the pod from starting (§3), so for {@code fetch} such an Error becomes "no cache, exit 0". Other
     * commands keep their normal behaviour, and {@code --strict} is the operator asking for a failure.
     */
    static int exitCode(String[] args, java.util.function.IntSupplier run) {
        try {
            return run.getAsInt();
        } catch (OutOfMemoryError | StackOverflowError error) {
            boolean fetch = args.length > 0 && args[0].equals("fetch");
            if (!fetch || java.util.Arrays.asList(args).contains("--strict")) {
                throw error;
            }
            System.err.println("fetch: " + error.getClass().getSimpleName() + "; the application will start without a cache");
            return 0;
        }
    }
